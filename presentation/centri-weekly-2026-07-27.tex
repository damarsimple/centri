\begin{frame}{A defect the checker missed}
\vfill
\begin{center}
\begin{tcolorbox}[colback=cAccent!6,colframe=cAccent,boxsep=2pt,left=8pt,right=8pt,top=5pt,bottom=5pt,width=0.92\textwidth]
\centering ``a full turn in \textbf{2.07 seconds}'' \dots\ ``\emph{that} turn rate is
\textbf{0.30 full turns each second}''\\[2pt]
{\footnotesize 0.30 turns each second \emph{is} a 3.34-second turn}
\end{tcolorbox}
\end{center}
\vfill
\begin{center}
\small Both are measured --- the fastest lap, and the clip average --- so the checker passed it.
\end{center}
\vfill
\begin{center}
{\footnotesize A new rule catches it, and agrees with the judge clip for clip:
the same six flagged, the same one clean.}
\end{center}
\takeaway{Instructing the writer to stop made every score worse --- the period and the rate have to
reach it already paired.}
\end{frame}
